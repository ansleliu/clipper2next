cmake_minimum_required(VERSION 3.24)
project(clipper2next_public_triangulation LANGUAGES CXX)
foreach(input PUBLIC_HEADERS PUBLIC_LIBRARY)
  if(NOT EXISTS "${${input}}")
    message(FATAL_ERROR "${input} must name the exact matching frozen public artifact")
  endif()
endforeach()
add_executable(clipper2next_public_triangulation main.cpp)
target_compile_features(clipper2next_public_triangulation PRIVATE cxx_std_23)
target_include_directories(clipper2next_public_triangulation PRIVATE "${PUBLIC_HEADERS}")
target_link_libraries(clipper2next_public_triangulation PRIVATE "${PUBLIC_LIBRARY}")
if(MSVC)
  target_compile_options(clipper2next_public_triangulation PRIVATE /W4 /showIncludes)
else()
  target_compile_options(clipper2next_public_triangulation PRIVATE -Wall -Wextra -Wpedantic)
  set_target_properties(clipper2next_public_triangulation PROPERTIES BUILD_RPATH "${PUBLIC_RUNTIME_DIR}")
endif()
